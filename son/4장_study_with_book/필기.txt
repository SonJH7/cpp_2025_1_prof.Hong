4.3 pointer 변수와 reference 변수
    reference 변수는 기존 변수에 대한 aliasing으로 같은 메모리 공간을 가리킨다.
    int j = 22;
    int &k = j;
    k++; //j 값이 23으로 증가한다.
    int &m; //error 발생, reference 변수는 항상 초기화해야 함
    int *const m = &j; //reference 변수 k는 *m과 같은 역할을 한다.

    int n = 55;
    int *p; //초기화 없어도 사용 가능
    p = &n;
    n++; (*p)++; //n의 값이 56, 57이 된다.
    *p++; //n의 값이 바뀌는 것이 아니다.
    int &q = n;//reference 변수는 항상 초기화해야 한다.
    n++; q++; //n의 값이 58, 59가 된다.
